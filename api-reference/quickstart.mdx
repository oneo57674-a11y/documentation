  <script>
    // Game State
    let score = 0;
    let level = 1;
    let currentLane = 1; // 0: Left, 1: Center, 2: Right
    let isPlaying = false;
    let wallPosition = -70;
    let wallSpeed = 3;
    let currentQuestion = null;
    let animationFrameId = null;

    const lanePositions = [16.66, 50, 83.33]; // Percentage positions for 3 lanes

    // DOM Elements
    const playerEl = document.getElementById('player');
    const wallGroupEl = document.getElementById('wall-group');
    const questionTextEl = document.getElementById('question-text');
    const scoreDisplay = document.getElementById('score-display');
    const levelDisplay = document.getElementById('level-display');
    const startScreen = document.getElementById('start-screen');
    const gameOverScreen = document.getElementById('game-over-screen');

    // Key Listener
    window.addEventListener('keydown', (e) => {
      if (!isPlaying) return;
      if (e.key === 'ArrowLeft' && currentLane > 0) {
        movePlayer(currentLane - 1);
      } else if (e.key === 'ArrowRight' && currentLane < 2) {
        movePlayer(currentLane + 1);
      }
    });

    function movePlayer(laneIndex) {
      currentLane = laneIndex;
      playerEl.style.left = `${lanePositions[currentLane]}%`;
    }

    function generateMathProblem(currentLevel) {
      let num1, num2, answer, operator;
      const operators = ['+'];
      
      if (currentLevel >= 3) operators.push('-');
      if (currentLevel >= 6) operators.push('×');

      operator = operators[Math.floor(Math.random() * operators.length)];

      const maxNum = Math.min(10 + currentLevel * 3, 100);

      if (operator === '+') {
        num1 = Math.floor(Math.random() * maxNum) + 1;
        num2 = Math.floor(Math.random() * maxNum) + 1;
        answer = num1 + num2;
      } else if (operator === '-') {
        num1 = Math.floor(Math.random() * maxNum) + 10;
        num2 = Math.floor(Math.random() * num1) + 1;
        answer = num1 - num2;
      } else if (operator === '×') {
        num1 = Math.floor(Math.random() * (5 + currentLevel)) + 2;
        num2 = Math.floor(Math.random() * 10) + 2;
        answer = num1 * num2;
      }

      // Generate wrong choices
      let choices = [answer];
      while (choices.length < 3) {
        let wrongAnswer = answer + (Math.floor(Math.random() * 10) - 5);
        if (wrongAnswer !== answer && wrongAnswer >= 0 && !choices.includes(wrongAnswer)) {
          choices.push(wrongAnswer);
        }
      }

      // Shuffle choices
      choices.sort(() => Math.random() - 0.5);

      return {
        text: `${num1} ${operator} ${num2} = ?`,
        answer: answer,
        correctLane: choices.indexOf(answer),
        choices: choices
      };
    }

    function resetWall() {
      wallPosition = -70;
      currentQuestion = generateMathProblem(level);
      
      questionTextEl.textContent = currentQuestion.text;
      
      for (let i = 0; i < 3; i++) {
        document.getElementById(`wall-${i}`).textContent = currentQuestion.choices[i];
      }
    }

    function startGame() {
      score = 0;
      level = 1;
      wallSpeed = 3;
      isPlaying = true;
      
      scoreDisplay.textContent = score;
      levelDisplay.textContent = level;
      
      movePlayer(1);
      startScreen.classList.add('hidden');
      gameOverScreen.classList.add('hidden');

      resetWall();
      if (animationFrameId) cancelAnimationFrame(animationFrameId);
      gameLoop();
    }

    function gameOver() {
      isPlaying = false;
      document.getElementById('final-score').textContent = score;
      document.getElementById('final-level').textContent = level;
      gameOverScreen.classList.remove('hidden');
    }

    function gameLoop() {
      if (!isPlaying) return;

      wallPosition += wallSpeed;
      wallGroupEl.style.top = `${wallPosition}px`;

      const trackHeight = document.getElementById('track').clientHeight;
      const collisionThreshold = trackHeight - 90; // Near player height

      // Check Collision
      if (wallPosition >= collisionThreshold && wallPosition <= collisionThreshold + wallSpeed) {
        if (currentLane === currentQuestion.correctLane) {
          // ตอบถูก
          score += 10;
          scoreDisplay.textContent = score;
          
          // เพิ่มระดับความยากทุก 30 คะแนน
          if (score % 30 === 0) {
            level++;
            wallSpeed += 0.5;
            levelDisplay.textContent = level;
          }

          resetWall();
        } else {
          // ตอบผิด - เกมจบ
          gameOver();
          return;
        }
      }

      // If wall passes bottom without collision fallback reset
      if (wallPosition > trackHeight) {
        resetWall();
      }

      animationFrameId = requestAnimationFrame(gameLoop);
    }
  </script>

This guide walks you through making your first API request to bunny.net.

<Steps>
  <Step title="Get your API key">
    1. Log in to the [bunny.net dashboard](https://dash.bunny.net)
    2. Navigate to **Account** > **API**
    3. Copy your **Account API Key**

    <Warning>
    Keep your API key secure and never share it publicly.
    </Warning>

  </Step>

  <Step title="Make your first request">
    Use the following curl command to list your pull zones:

    ```bash
    curl --request GET \
      --url https://api.bunny.net/pullzone \
      --header 'AccessKey: YOUR_API_KEY' \
      --header 'Content-Type: application/json'
    ```

    Replace `YOUR_API_KEY` with the API key you copied from the dashboard.

  </Step>

  <Step title="View the response">
    A successful response returns a list of your pull zones:

    ```json
    {
      "Items": [
        {
          "Id": 1234,
          "Name": "my-pull-zone",
          "OriginUrl": "https://example.com",
          "Enabled": true,
          "Hostnames": [
            {
              "Id": 5678,
              "Value": "my-pull-zone.b-cdn.net"
            }
          ]
        }
      ],
      "CurrentPage": 0,
      "TotalItems": 1,
      "HasMoreItems": false
    }
    ```

    If you don't have any pull zones yet, the `Items` array will be empty.

  </Step>
